cmake_minimum_required(VERSION 3.20)
project(aips_signal VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(AIPS_ENABLE_CUDA "Build the CUDA batch reduction kernel" OFF)
option(AIPS_BUILD_TESTS "Build unit tests" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Path to an ONNX Runtime release (include/ and lib/)")

add_library(aips_signal STATIC src/signal.cpp src/batch.cpp src/rul.cpp)
target_include_directories(aips_signal PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(aips_signal PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic -Wconversion -Wno-sign-conversion -O2>)
find_package(Threads REQUIRED)
target_link_libraries(aips_signal PUBLIC Threads::Threads)

if(ONNXRUNTIME_ROOT AND EXISTS "${ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
  message(STATUS "aips-signal: ONNX Runtime at ${ONNXRUNTIME_ROOT}")
  target_compile_definitions(aips_signal PUBLIC AIPS_WITH_ONNXRUNTIME=1)
  target_include_directories(aips_signal PRIVATE ${ONNXRUNTIME_ROOT}/include)
  target_link_directories(aips_signal PUBLIC ${ONNXRUNTIME_ROOT}/lib)
  target_link_libraries(aips_signal PUBLIC onnxruntime)
else()
  message(WARNING "aips-signal: building WITHOUT ONNX Runtime; RUL inference disabled")
endif()

if(AIPS_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    target_sources(aips_signal PRIVATE src/batch_cuda.cu)
    target_compile_definitions(aips_signal PUBLIC AIPS_ENABLE_CUDA=1)
    set_target_properties(aips_signal PROPERTIES CUDA_ARCHITECTURES "75;86;89")
  else()
    message(WARNING "aips-signal: AIPS_ENABLE_CUDA=ON but no CUDA compiler found; CPU only")
  endif()
endif()

if(AIPS_BUILD_TESTS)
  enable_testing()
  add_executable(aips_signal_tests tests/test_signal.cpp)
  target_link_libraries(aips_signal_tests PRIVATE aips_signal)
  add_test(NAME aips_signal_tests COMMAND aips_signal_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../ml/models/rul_qr.onnx)
endif()

install(TARGETS aips_signal ARCHIVE DESTINATION lib)
install(FILES include/aips_signal.h DESTINATION include)
